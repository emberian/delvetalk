def drop(state: State, context: Abi.Context) -> Activity<Nat>:
  let written(_) = write {coins: insert {n: Relations.count(state.coins), who: context.principal}}
  Relations.count(state.coins) + 1n

  Document.text("{natText(Relations.count(state.coins))} coins\n")

  Card.answer(input, context, forms(), render(state, context))
