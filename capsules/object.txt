import ./Abi.obend as Abi
import ./List.obend as Lists
import ./Document.obend as Document
import ./Relation.obend as Relations
import ./Text.obend as Text
record Coin:
  n: Nat
  who: String
  line: String
record State:
  owner: String
  coins: Relations.Relation<Coin>
def initial() -> State:
  {owner: "", coins: Relations.empty()}
def relations() -> Relations.Decls:
  Lists.append(Relations.Decls.nil(), {field: "coins", key: Text.words("n who"), limit: 64n})
law owner "anyone drops; the owner reprograms": request.kind == 0 or request.subject == new.owner
form drop:
  line: text 1..140
def drop(state: State, input: {line: String}, context: Abi.Context) -> Activity<Nat>:
  let written(_) = write {coins: insert {n: Relations.count(state.coins), who: context.principal, line: input.line}}
  Relations.count(state.coins) + 1n
def render(state: State, context: Abi.Context) -> Document.Document:
  Document.text("{natText(Relations.count(state.coins))} coins\n")
